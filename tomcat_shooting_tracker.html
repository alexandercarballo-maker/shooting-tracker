Google Apps Script (doPost) 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Shooting Log") || ss.insertSheet("Shooting Log");
  var headers = ["Timestamp", "Workout", "Date", "Formatted Date", "Drill", "Makes", "Misses", "Attempts", "Percentage", "Knee Comfort", "Notes"];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
  }
  var data = JSON.parse(e.postData.contents);
  var rows = data.rows || [];
  var existing = sheet.getDataRange().getValues();

  rows.forEach(function(r) {
    var misses = Math.max(0, Number(r.attempts || 0) - Number(r.makes || 0));
    var rowValues = [
      r.timestamp, r.workout, r.weekOf, r.day,
      r.drill, Number(r.makes || 0), misses, Number(r.attempts || 0), r.percentage,
      r.kneeComfort, r.notes
    ];
    var updated = false;
    for (var i = 1; i < existing.length; i++) {
      if (String(existing[i][1]) === String(r.workout) &&
          String(existing[i][2]) === String(r.weekOf) &&
          String(existing[i][4]) === String(r.drill)) {
        sheet.getRange(i + 1, 1, 1, rowValues.length).setValues([rowValues]);
        updated = true;
        break;
      }
    }
    if (!updated) {
      sheet.appendRow(rowValues);
    }
  });
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", count: rows.length }))
    .setMimeType(ContentService.MimeType.JSON);
}
