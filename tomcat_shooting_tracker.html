Google Apps Script (doPost) 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Shooting Log") || ss.insertSheet("Shooting Log");
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Timestamp", "Workout", "Week Of", "Day", "Drill", "Makes", "Attempts", "Percentage", "Knee Comfort", "Notes"]);
  }
  var data = JSON.parse(e.postData.contents);
  var rows = data.rows || [];
  rows.forEach(function(r) {
    sheet.appendRow([
      r.timestamp, r.workout, r.weekOf, r.day,
      r.drill, r.makes, r.attempts, r.percentage,
      r.kneeComfort, r.notes
    ]);
  });
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", count: rows.length }))
    .setMimeType(ContentService.MimeType.JSON);
}
