Google Sheets Sync Setup

Paste your Google Apps Script Web App URL below so you can Push sessions from your iPhone and Pull all sessions onto your laptop anytime.

Apps Script Code (Supports Push & Pull) 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Shooting Log") || ss.insertSheet("Shooting Log");
  var headers = ["Timestamp", "Workout", "Date", "Formatted Date", "Drill", "Makes", "Misses", "Attempts", "Percentage", "Knee Comfort", "Notes", "Spot Breakdown"];
  if (sheet.getLastRow() === 0) sheet.appendRow(headers);
  var data = JSON.parse(e.postData.contents);
  var rows = data.rows || [];
  var existing = sheet.getDataRange().getValues();
  rows.forEach(function(r) {
    var misses = Math.max(0, Number(r.attempts || 0) - Number(r.makes || 0));
    var rowValues = [
      r.timestamp, r.workout, r.weekOf, r.day,
      r.drill, Number(r.makes || 0), misses, Number(r.attempts || 0), r.percentage,
      r.kneeComfort, r.notes, r.spotBreakdown || ""
    ];
    var updated = false;
    for (var i = 1; i < existing.length; i++) {
      if (String(existing[i][1]) === String(r.workout) &&
          String(existing[i][2]) === String(r.weekOf) &&
          String(existing[i][4]) === String(r.drill)) {
        sheet.getRange(i + 1, 1, 1, rowValues.length).setValues([rowValues]);
        updated = true;
        break;
      }
    }
    if (!updated) sheet.appendRow(rowValues);
  });
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", count: rows.length }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Shooting Log");
  if (!sheet || sheet.getLastRow() < 2) {
    return ContentService.createTextOutput(JSON.stringify({ status: "ok", rows: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var values = sheet.getDataRange().getValues();
  var out = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    out.push({
      timestamp: row[0],
      workout: String(row[1] || ""),
      date: String(row[2] || ""),
      formattedDate: String(row[3] || ""),
      drill: String(row[4] || ""),
      makes: Number(row[5] || 0),
      misses: Number(row[6] || 0),
      attempts: Number(row[7] || 0),
      percentage: String(row[8] || ""),
      kneeComfort: String(row[9] || "Great"),
      notes: String(row[10] || ""),
      spotBreakdown: String(row[11] || "")
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", rows: out }))
    .setMimeType(ContentService.MimeType.JSON);
}
